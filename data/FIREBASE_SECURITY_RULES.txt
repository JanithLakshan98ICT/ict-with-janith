rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
    match /admins/{uid} { allow read: if signedIn() && request.auth.uid == uid; allow write: if false; }
    match /students/{uid} { allow read: if isAdmin() || (signedIn() && request.auth.uid == uid); allow create,update,delete: if isAdmin(); }
    match /results/{id} { allow read: if isAdmin() || (signedIn() && resource.data.studentUid == request.auth.uid); allow create,update,delete: if isAdmin(); }
    match /exams/{id} { allow read: if signedIn(); allow create,update,delete: if isAdmin(); }
    match /resources/{id} { allow read: if signedIn() && resource.data.active == true; allow create,update,delete: if isAdmin(); }
    match /announcements/{id} { allow read: if signedIn() && resource.data.active == true; allow create,update,delete: if isAdmin(); }
  }
}
