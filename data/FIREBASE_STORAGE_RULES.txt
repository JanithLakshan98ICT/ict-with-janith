rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {
    // Student profile photos are displayed in the student profile and birthday post.
    match /student-photos/{uid}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null
        && exists(/databases/(default)/documents/admins/$(request.auth.uid))
        && get(/databases/(default)/documents/admins/$(request.auth.uid)).data.active == true
        && request.resource.size < 3 * 1024 * 1024
        && request.resource.contentType.matches('image/.*');
    }
  }
}
